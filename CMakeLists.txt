cmake_minimum_required(VERSION 3.22)
project(tgit2 VERSION 0.2.0 LANGUAGES C)
option(BUILD_SHARED_LIBS "Build a loadable Tcl extension" ON)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
find_package(TclStub 8.6 REQUIRED)
find_package(libgit2 1.9 CONFIG REQUIRED)

add_library(tgit2 src/tgit2.c src/workspace.c)
set_target_properties(tgit2 PROPERTIES PREFIX "lib" POSITION_INDEPENDENT_CODE ON C_STANDARD 11)
target_include_directories(tgit2 PRIVATE "${TCL_INCLUDE_PATH}")
target_link_libraries(tgit2 PRIVATE libgit2::libgit2package)
target_compile_definitions(tgit2 PRIVATE PACKAGE_VERSION="${PROJECT_VERSION}")
set(pkgdir lib/tgit2${PROJECT_VERSION})
install(TARGETS tgit2 DESTINATION ${pkgdir})
if(BUILD_SHARED_LIBS)
    target_compile_definitions(tgit2 PRIVATE USE_TCL_STUBS)
    target_link_libraries(tgit2 PRIVATE "${TCL_STUB_LIBRARY}")
    if(APPLE)
        set_target_properties(tgit2 PROPERTIES INSTALL_RPATH "@loader_path/..")
    elseif(UNIX)
        set_target_properties(tgit2 PROPERTIES INSTALL_RPATH "$ORIGIN/..")
    endif()
    configure_file(pkgIndex.tcl.in pkgIndex.tcl @ONLY)
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/pkgIndex.tcl" DESTINATION ${pkgdir})
    include(CTest)
    if(BUILD_TESTING)
        add_test(NAME clone COMMAND "${TCL_TCLSH}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/clone.test")
        add_test(NAME workspace COMMAND "${TCL_TCLSH}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/workspace.test")
        set_tests_properties(clone workspace PROPERTIES ENVIRONMENT "TCLLIBPATH=${CMAKE_CURRENT_BINARY_DIR}")
    endif()
else()
    target_compile_definitions(tgit2 PRIVATE STATIC_BUILD)
endif()
